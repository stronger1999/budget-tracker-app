import {createSlice,PayloadAction} from '@reduxjs/toolkit';import {User} from '../../types';import {load,save} from '../../app/storage';
const demo:User={id:'demo',name:'Student',email:'demo@budget.app',password:'password123'};type State={users:User[];currentUser:Omit<User,'password'>|null;error:string|null};
const initial:State=load('bt_auth',{users:[demo],currentUser:null,error:null});
const slice=createSlice({name:'auth',initialState:initial,reducers:{register:(s,a:PayloadAction<{name:string;email:string;password:string}>)=>{if(s.users.some(u=>u.email.toLowerCase()===a.payload.email.toLowerCase())){s.error='Account already exists';return}const u={id:crypto.randomUUID(),...a.payload};s.users.push(u);s.currentUser={id:u.id,name:u.name,email:u.email};s.error=null;save('bt_auth',s)},login:(s,a:PayloadAction<{email:string;password:string}>)=>{const u=s.users.find(x=>x.email.toLowerCase()===a.payload.email.toLowerCase()&&x.password===a.payload.password);if(!u){s.error='Invalid email or password';return}s.currentUser={id:u.id,name:u.name,email:u.email};s.error=null;save('bt_auth',s)},logout:(s)=>{s.currentUser=null;save('bt_auth',s)},clearAuthError:(s)=>{s.error=null}}});
export const {register,login,logout,clearAuthError}=slice.actions;export default slice.reducer;
